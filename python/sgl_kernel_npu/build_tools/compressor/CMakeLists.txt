cmake_minimum_required(VERSION 3.20)
project(sgl_compressor LANGUAGES CXX)

find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c
    "import pathlib, torch, torch_npu; print(torch.utils.cmake_prefix_path); print(pathlib.Path(torch_npu.__file__).parent)"
    RESULT_VARIABLE python_result OUTPUT_VARIABLE python_paths OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT python_result EQUAL 0)
    message(FATAL_ERROR "The build Python must provide torch and torch_npu")
endif()
string(REPLACE "\n" ";" python_paths "${python_paths}")
list(GET python_paths 0 torch_prefix)
list(GET python_paths 1 torch_npu_root)
list(PREPEND CMAKE_PREFIX_PATH "${torch_prefix}")
find_package(Torch REQUIRED)
find_library(TORCH_NPU_LIBRARY torch_npu PATHS "${torch_npu_root}/lib" NO_DEFAULT_PATH REQUIRED)
set(sdk "$ENV{ASCEND_HOME_PATH}")
set(ASCEND_HOME_PATH "${sdk}")
set(ASCEND_CANN_PACKAGE_PATH "${sdk}")
foreach(candidate IN ITEMS tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake ascendc_devkit/tikcpp/samples/cmake)
    if(EXISTS "${sdk}/${candidate}/ascendc.cmake")
        set(ASCENDC_CMAKE_DIR "${sdk}/${candidate}")
        break()
    endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "Cannot find ascendc.cmake in the initialized SDK")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
if(SGL_ARCH_VERSION STREQUAL "2201")
    set(arch arch22)
    set(arch_definition SGL_KERNEL_ARCH_22)
elseif(SGL_ARCH_VERSION STREQUAL "3510")
    set(arch arch35)
    set(arch_definition SGL_KERNEL_ARCH_35)
else()
    message(FATAL_ERROR "Compressor requires ARCH 2201 or 3510")
endif()
set(root "${SGL_SOURCE_ROOT}/csrc")
file(READ "${root}/pytorch_extensions.cpp" declarations)
string(REGEX MATCH "m\\.def\\([ \t\r\n]*\"compressor\\([^;]*;" schema "${declarations}")
if(NOT schema)
    message(FATAL_ERROR "Cannot find the original compressor schema")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registration.cpp"
    "#include <ATen/ATen.h>\n#include <torch/library.h>\n#include \"sgl_kenel_npu_ops.h\"\n"
    "TORCH_LIBRARY_FRAGMENT(npu, m) {\n${schema}\n}\n"
    "TORCH_LIBRARY_IMPL(npu, PrivateUse1, m) {\n"
    "m.impl(\"compressor\", TORCH_FN(sglang::npu_kernel::compressor));\n}\n")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
ascendc_library(compressor_kernel STATIC "${root}/compressor/op_kernel/compressor.cpp")
ascendc_include_directories(compressor_kernel PRIVATE
    "${root}/compressor/op_kernel" "${root}/compressor/op_kernel/${arch}" "${root}/utils/kernel")
ascendc_compile_definitions(compressor_kernel PRIVATE -DHAVE_WORKSPACE -DHAVE_TILING)
add_library(sgl_compressor SHARED "${CMAKE_CURRENT_BINARY_DIR}/registration.cpp"
    "${root}/compressor/op_host/compressor.cpp" "${root}/compressor/op_host/${arch}/compressor_tiling.cpp")
target_compile_features(sgl_compressor PRIVATE cxx_std_17)
get_target_property(torch_cxx_standard torch CXX_STANDARD)
if(torch_cxx_standard)
    set_property(TARGET sgl_compressor PROPERTY CXX_STANDARD "${torch_cxx_standard}")
endif()
separate_arguments(torch_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(sgl_compressor PRIVATE ${torch_flags})
target_compile_definitions(sgl_compressor PRIVATE ${arch_definition})
target_include_directories(sgl_compressor PRIVATE "${SGL_SOURCE_ROOT}/include"
    "${root}/utils" "${torch_npu_root}/include" ${Python3_INCLUDE_DIRS}
    "${sdk}/include" "${sdk}/include/external" "${sdk}/include/experiment/platform"
    "${sdk}/include/experiment/runtime")
target_link_libraries(sgl_compressor PRIVATE compressor_kernel ${TORCH_LIBRARIES} "${TORCH_NPU_LIBRARY}" ${CMAKE_DL_LIBS})
foreach(name IN ITEMS ascendcl tiling_api register platform ascendalog)
    find_library(${name}_library NAMES ${name} PATHS "${sdk}/lib64" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(sgl_compressor PRIVATE "${${name}_library}")
endforeach()
find_package(OpenMP REQUIRED COMPONENTS CXX)
target_link_libraries(sgl_compressor PRIVATE OpenMP::OpenMP_CXX)
target_link_options(sgl_compressor PRIVATE "-Wl,-Bsymbolic-functions")
set_target_properties(sgl_compressor PROPERTIES CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON SKIP_BUILD_RPATH ON INSTALL_RPATH ""
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
