function(npu_op_code_gen)
    cmake_parse_arguments(PARSE_ARGV 0 codegen "" "PACKAGE;OUT_DIR" "SRC;COMPILE_OPTIONS")
    if(NOT codegen_SRC STREQUAL expected_host)
        message(FATAL_ERROR "Codegen must use the declared CANN Host source")
    endif()
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/operator-config/sgl_operator_config.h" config)
    if(NOT config STREQUAL "#pragma once\n#define SGL_CANN_SOC_CONFIG \"${SGL_CANN_SOC}\"\n")
        message(FATAL_ERROR "OpDef configuration must use the selected build target")
    endif()
    if(NOT IS_DIRECTORY "${codegen_OUT_DIR}")
        message(FATAL_ERROR "Codegen output directory must exist before invoking the SDK")
    endif()
    file(WRITE "${codegen_OUT_DIR}/aclnn_fixture.h" "void aclnn${operator_type}GetWorkspaceSize();\n")
endfunction()

endfunction()

function(npu_op_kernel_sources target)
    cmake_parse_arguments(PARSE_ARGV 1 mapping ""
        "OP_TYPE;KERNEL_DIR;KERNEL_FILE" "COMPUTE_UNIT;SOC_SERIES")
    if(mapping_UNPARSED_ARGUMENTS OR mapping_KEYWORDS_MISSING_VALUES)
        message(FATAL_ERROR "Invalid Kernel source mapping arguments")
    endif()
    if(DEFINED mapping_COMPUTE_UNIT OR DEFINED mapping_SOC_SERIES)
        message(FATAL_ERROR "Shared Kernel source mapping must not add a device filter")
    endif()
    get_filename_component(entry "${mapping_KERNEL_DIR}/${mapping_KERNEL_FILE}"
        ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT target STREQUAL "sgl_cann_kernels"
       OR NOT mapping_OP_TYPE STREQUAL operator_type
       OR NOT entry STREQUAL expected_kernel)
        message(FATAL_ERROR "Kernel mapping does not resolve to the declared CANN entry")
    endif()
    set(kernel_mapping_seen ON PARENT_SCOPE)
endfunction()

function(npu_op_kernel_library target)
    cmake_parse_arguments(PARSE_ARGV 1 library "" "SRC_BASE;TILING_LIBRARY" "")
    if(NOT kernel_mapping_seen
       OR NOT library_SRC_BASE STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}"
       OR NOT library_TILING_LIBRARY STREQUAL "sgl_cann_tiling")
        message(FATAL_ERROR "Kernel library must use the registered source mapping and Tiling")
    endif()
endfunction()
